           """Reload _TAG_CACHE from app.session.workspace_path's vocabulary.

           try:
               vocab = load_vocabulary(app.session.workspace_path)
           except VocabularyError:
               # No vocab yet → empty cache, completer falls back to no suggestions
               for k in _TAG_CACHE:
                   _TAG_CACHE[k].clear()
               return
           cat_map = {
               TagCategory.PHYSICS_MODEL: "physics_model",
               TagCategory.MATH_TECHNIQUE: "math_technique",
               TagCategory.HEURISTIC: "heuristic",
           }
           for k in _TAG_CACHE:
               _TAG_CACHE[k].clear()
           for internal_id, tag in vocab.tags.items():
               key = cat_map.get(tag.category)
               if key is not None:
                   _TAG_CACHE[key][internal_id] = tag.display_zh

    5. SearchCommandSet class:
       ```python
       class SearchCommandSet(cmd2.CommandSet):
           @with_argparser(_search_parser)
           def do_search(self, ns) -> None:
               session = self._cmd.session
               if session.index_meta is None:
                   self._cmd.perror(INDEX_MISSING_MSG.format(workspace=session.workspace_path))
                   return
               def _split(arg: str | None) -> list[str] | None:
                   if arg is None: return None
                   return [t.strip() for t in arg.split(",") if t.strip()]
               try:
                   results = query_index(
                       session.workspace_path,
                       physics_model_ids=_split(ns.physics_model),
                       math_technique_ids=_split(ns.math_technique),
                       heuristic_ids=_split(ns.heuristic),
                       match_mode=ns.match_mode,
                   )
               except IndexNotFoundError:
                   self._cmd.perror(INDEX_MISSING_MSG.format(workspace=session.workspace_path))
                   return
               # Truncate to max_results
               displayed = results[: session.max_results]
               # Update session state
               session.last_search_query = self._serialize_query(ns)
               session.last_search_result_ids = [e.problem_id for e in displayed]
               # Render table (use SimpleTable or fallback to plain rows)
               if not displayed:
                   self._cmd.poutput("未找到匹配题目。")
                   return
               self._cmd.poutput(self._render_table(displayed))
               if len(results) > session.max_results:
                   self._cmd.poutput(f"... 仅显示前 {session.max_results} 道，共 {len(results)} 道。调整 `set max_results N` 后重试。")
               else:
                   self._cmd.poutput(f"共 {len(displayed)} 道题目。")

           @with_argparser(_show_parser)
           def do_show(self, ns) -> None:
               session = self._cmd.session
               if session.index_meta is None:
                   self._cmd.perror(INDEX_MISSING_MSG.format(workspace=session.workspace_path))
                   return
               # Resolve ref → problem_id
               problem_id = self._resolve_ref(ns.ref, session)
               if problem_id is None:
                   return  # already perror'd
               try:
                   entry = get_problem_entry(session.workspace_path, problem_id)
               except IndexNotFoundError:
                   self._cmd.perror(INDEX_MISSING_MSG.format(workspace=session.workspace_path))
                   return
               if entry is None:
                   self._cmd.perror(f"索引中未找到题目: {problem_id}")
                   return
               session.current_problem_id = problem_id
               if ns.full or session.output_format == "full":
                   self._cmd.ppaged(self._render_full(entry))
               else:
                   self._cmd.poutput(self._render_compact(entry))
       ```
